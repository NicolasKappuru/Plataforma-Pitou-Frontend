"""
CREAR IMÁGENES (1-5 ARCHIVOS)
------------------------------
POST /api/plugins/gestor_imagenes/imagen/

Headers:
  Authorization: Bearer <jwt_token>
  Content-Type: multipart/form-data

Form Data (multipart/form-data):
  concepto: "1"                                    (requerido, ID del concepto)
  imagenes: <archivo1.png>                         (requerido, 1-5 archivos)
  imagenes: <archivo2.jpg>
  nombre_plugin: "Diagrama 1"                      (opcional, por cada archivo)
  nombre_plugin: "Diagrama 2"
  descripcion_plugin: "Descripción 1"              (opcional)
  descripcion_plugin: "Descripción 2"
  posicion: "0"                                    (opcional, por defecto índice)
  posicion: "1"

Ejemplo cURL:
curl -X POST "http://localhost:8000/api/plugins/gestor_imagenes/imagen/" \
  -H "Authorization: Bearer <jwt_token>" \
  -F "concepto=1" \
  -F "imagenes=@/ruta/imagen1.png" \
  -F "imagenes=@/ruta/imagen2.jpg" \
  -F "nombre_plugin=Diagrama 1" \
  -F "nombre_plugin=Diagrama 2" \
  -F "descripcion_plugin=Desc 1" \
  -F "descripcion_plugin=Desc 2"

Response 201:
{
  "imagenes": [
    {
      "id": 10,
      "nombre_plugin": "Diagrama 1",
      "descripcion_plugin": "Desc 1",
      "posicion": 0,

      "concepto": 1
    },
    {
      "id": 11,
      "nombre_plugin": "Diagrama 2",
      "descripcion_plugin": "Desc 2",
      "posicion": 1,
      "url_imagen": "https://supabase.co/storage/v1/object/public/bucket/imagenes/uuid2.jpg",
      "concepto": 1
    }
  ]
}

Response 400: {"error": "Debe enviar al menos una imagen"} / {"error": "Debe enviar entre 1 y 5 imágenes"}
Response 403: {"error": "No tiene permiso para crear"}
"""

# RESUMEN DE CÓDIGOS DE ESTADO COMUNES

"""
200 OK          - Consulta/Actualización exitosa
201 Created     - Creación exitosa
400 Bad Request - Datos inválidos, faltantes o límite excedido (max 5 items)
401 Unauthorized- Token JWT inválido o expirado
403 Forbidden   - Sin permiso requerido (consultar/crear/actualizar/eliminar)
404 Not Found   - Recurso no encontrado
500 Internal    - Error del servidor

LÍMITES:
- Máximo 5 items por request (crear/actualizar)
- Archivos de imagen: máximo 5, tipos permitidos según configuración Supabase
"""

# =============================================================================
# EJEMPLOS DE PERMISOS REQUERIDOS
# =============================================================================
"""
Todas las vistas usan el decorador @requiere_permiso('permiso'):

GET    -> requiere_permiso('consultar')
POST   -> requiere_permiso('crear')
PUT    -> requiere_permiso('actualizar')
DELETE -> requiere_permiso('eliminar')

Roles y permisos por defecto:
- administrador: todos los permisos
- autor: consultar, crear, actualizar, eliminar (sus propios recursos)
- lector: solo consultar
"""
